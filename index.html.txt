    function showTab(name, btn) {

    function formatDate(str) {
      if (!str) return '–';
      return new Date(str).toLocaleDateString('de-DE', {weekday:'short',day:'2-digit',month:'2-digit',year:'numeric'});
    }
    function formatTime(str) {
      if (!str) return '';
      return new Date(str).toLocaleTimeString('de-DE', {hour:'2-digit',minute:'2-digit'}) + ' Uhr';
    }
    function getResult(match, id1860) {
      const r = match.MatchResults?.find(x => x.ResultTypeID === 2) || match.MatchResults?.[match.MatchResults.length-1];
      if (!r) return 'upcoming';
      const home = match.Team1.TeamId === id1860;
      const g = home ? r.PointsTeam1 : r.PointsTeam2;
      const o = home ? r.PointsTeam2 : r.PointsTeam1;
      return g > o ? 'win' : g < o ? 'loss' : 'draw';
    }
    const SEASON_YEAR = new Date().getMonth() >= 6 ? new Date().getFullYear() : new Date().getFullYear() - 1;
    async function fetchMatches() {
      try {
        const res = await fetch(`https://api.openligadb.de/getmatchdata/bl3/${SEASON_YEAR}`);
        if (!res.ok) throw new Error();
        const data = await res.json();
        const games = data.filter(m =>
          [m.Team1.TeamName, m.Team2.TeamName, m.Team1.ShortName, m.Team2.ShortName]
            .some(n => n && (n.includes('1860') || n.toLowerCase().includes('sechzig')))
        );
        if (!games.length) { showFallback(); return; }
        const g0 = games[0];
        const id1860 = ([g0.Team1.TeamName, g0.Team1.ShortName].some(n => n?.includes('1860')))
          ? g0.Team1.TeamId : g0.Team2.TeamId;
        const played = games.filter(m => m.MatchIsFinished).slice(-6).reverse();
        const upcoming = games.filter(m => !m.MatchIsFinished).slice(0, 6);
        const rc = document.getElementById('results-container');
        if (!played.length) { rc.innerHTML = '<div class="error-msg">Noch keine Spiele absolviert.</div>'; }
        else rc.innerHTML = `<div class="results-grid">${played.map(m => {
          const r = m.MatchResults?.find(x => x.ResultTypeID===2) || m.MatchResults?.[m.MatchResults.length-1];
          const score = r ? `${r.PointsTeam1} : ${r.PointsTeam2}` : '– : –';
          const res = getResult(m, id1860);
          const cls = res==='win'?'result-win':res==='loss'?'result-loss':'result-draw';
          return `<div class="result-card ${cls}">
            <div class="result-date">${formatDate(m.MatchDateTimeUTC)} · ${formatTime(m.MatchDateTimeUTC)}</div>
            <div class="result-teams">
              <div class="result-team home">${m.Team1.ShortName||m.Team1.TeamName}</div>
              <div class="result-score">${score}</div>
              <div class="result-team away">${m.Team2.ShortName||m.Team2.TeamName}</div>
            </div>
            <div class="result-league">3. Liga · Spieltag ${m.Group?.GroupOrderID||''}</div>
          </div>`;
        }).join('')}</div>`;
        const uc = document.getElementById('upcoming-container');
        if (!upcoming.length) { uc.innerHTML = '<div class="error-msg">Keine weiteren Spiele geplant.</div>'; }
        else uc.innerHTML = `<div class="results-grid">${upcoming.map(m => `
          <div class="result-card" style="border-color:rgba(0,102,204,0.3);background:rgba(0,51,102,0.15)">
            <div class="result-date">Spieltag ${m.Group?.GroupOrderID||''}</div>
            <div class="result-teams">
              <div class="result-team home">${m.Team1.ShortName||m.Team1.TeamName}</div>

            </div>
            <div class="result-league">📅 ${formatDate(m.MatchDateTimeUTC)} · ⏰ ${formatTime(m.MatchDateTimeUTC)}</div>
          </div>`).join('')}</div>`;
      } catch(e) { showFallback(); }
    }
